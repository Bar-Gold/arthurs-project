# Only the Playwright driver is needed, not its bundled browsers: the app
# attaches to the user's real Chrome and never launches one itself.
playwright>=1.62
pytest>=9.1

# The UI. Qt rather than Tk because Tk 8.6 has no bidirectional text support:
# it lays characters out in logical order and leaves reordering to Windows,
# which only manages one run of one script at a time, so any line mixing Hebrew
# with English or digits came out mirrored. Qt shapes text itself and needs no
# help at all. LGPL, so PySide6 rather than PyQt6.
PySide6>=6.6

# Tests only: tests/test_qtui_preview.py writes its test images with it, and
# skips without it. The app decodes images with Qt and never imports it.
Pillow>=10.0

# Windows ships no IANA time zone database, so Asia/Jerusalem is unavailable
# without this. The posting window is judged in Israel local time, so it is a
# hard requirement rather than a nicety.
tzdata
